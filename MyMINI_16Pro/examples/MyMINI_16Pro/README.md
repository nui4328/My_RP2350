void sensor()
  {
    const uint8_t channel = 0; // ใช้เซนเซอร์ F0/B0 ในตัวอย่าง
    readADC_F(channel);  // อ่านค่า F0 ที่แปลงเป็น 0–1000
    minADC_F(channel);   // ค่าต่ำสุดของสเกล F0 = 0 เมื่อคาลิเบรตแล้ว
    maxADC_F(channel);   // ค่าสูงสุดของสเกล F0 = 1000 เมื่อคาลิเบรตแล้ว
    readADC_B(channel);  // อ่านค่า B0 ที่แปลงเป็น 0–1000
    minADC_B(channel);   // ค่าต่ำสุดของสเกล B0 = 0 เมื่อคาลิเบรตแล้ว
    maxADC_B(channel);   // ค่าสูงสุดของสเกล B0 = 1000 เมื่อคาลิเบรตแล้ว
    readADC_CL();        // อ่านค่ากลางซ้าย CL ที่แปลงเป็น 0–1000
    minADC_CL();         // ค่าต่ำสุดของสเกล CL = 0 เมื่อคาลิเบรตแล้ว
    maxADC_CL();         // ค่าสูงสุดของสเกล CL = 1000 เมื่อคาลิเบรตแล้ว
    readADC_CR();        // อ่านค่ากลางขวา CR ที่แปลงเป็น 0–1000
    minADC_CR();         // ค่าต่ำสุดของสเกล CR = 0 เมื่อคาลิเบรตแล้ว
    maxADC_CR();         // ค่าสูงสุดของสเกล CR = 1000 เมื่อคาลิเบรตแล้ว

    int32_t aux = readADC_F();  // ADS1115 AIN3: ค่าดิบ (ยังไม่คาลิเบรตเป็น 0–1000)
    if (aux != ADC_VALUE_INVALID) {
      Serial.print("AIN3 raw: "); Serial.println(aux);
    } else {
      Serial.println("AIN3 read failed");
    }

    read_rowSensor();
  }

void gyro_mini()
  {
    float yaw, rateX, rateY, rateZ;
    if (!gyro_read(yaw, rateX, rateY, rateZ)) return; // อ่านหัวและอัตราหมุน; อ่านไม่ได้ให้ออก

    Serial.print(rateX, 2); Serial.print('/');
    Serial.print(rateY, 2); Serial.print('/');
    Serial.println(rateZ, 2);

    Serial.print(F("Yaw after reset: "));
    Serial.println(yaw, 2);
  }

## คาลิเบรต BNO055 ที่ใช้กับหุ่นยนต์

หลัง `robot_setup()` และก่อนเริ่มภารกิจ วางหุ่นยนต์นิ่ง แล้วเรียก:

```cpp
if (gyro_calibrate(true)) {
  Serial.println("Gyro offset บันทึกและตรวจสอบแล้ว");
} else {
  Serial.println("Gyro calibration ไม่สำเร็จ");
}
```

`gyro_calibrate(true)` บังคับเก็บ gyro offset ใหม่ หากต้องการโหลดค่าเดิมก่อนและเก็บใหม่เฉพาะเมื่อไม่มีค่า ใช้ `gyro_calibrate(false)` ฟังก์ชันหยุดมอเตอร์ เตรียม BNO055 ตรวจตัวอย่างขณะนิ่ง บันทึกและอ่าน EEPROM กลับ จากนั้นตั้งมุมอ้างอิงใหม่ คืน `false` เมื่อไม่พร้อม เคลื่อนระหว่างเก็บ หรือบันทึก/อ่านยืนยันไม่สำเร็จ เฉพาะ BNO055 ไม่รองรับ BNO085

ครั้งต่อไปเมื่อ `gyro_reset()` / `gyro_set()` / `wait_button()` เริ่ม BNO055 ไลบรารีจะพยายามโหลด gyro offset เดิมจาก EEPROM โดยไม่เก็บใหม่อัตโนมัติ ตรวจสถานะด้วย `gyro_offset_calibrated()`; ถ้าไม่มีค่าเดิมยังใช้ไจโรต่อได้ตามปกติ

offset นี้ใช้แก้ค่าอัตราหมุน X/Y/Z ที่ไลบรารีอ่าน รวมถึงตรวจความสมเหตุสมผลของ heading; มุม heading ยังมาจาก fusion ของ BNO055 ไม่ใช่การนำ offset นี้ไปหักจากมุมโดยตรง การบันทึก fusion profile ด้วย `save_gyro_calibration()` เป็นอีกกระบวนการหนึ่ง

ใน `MyMINI_16Pro.ino` มีบรรทัด `gyro_calibrate(true)` ที่คอมเมนต์ไว้ ปลดคอมเมนต์เมื่อจะคาลิเบรตใหม่ แล้วใส่คอมเมนต์กลับหลังบันทึกสำเร็จ เพื่อไม่เขียน EEPROM ซ้ำทุกครั้งที่เปิดเครื่อง

เมื่อคาลิเบรตสำเร็จและตรวจสอบค่าที่บันทึกแล้ว จะมีเสียงเมโลดีสั้น 3 โน้ต เสียงเดียวกันดังเมื่อเข้า `wait_button()` แล้วพบ BNO055 พร้อมค่า gyro offset ที่โหลดไว้ หากไม่พบค่า หรือคาลิเบรตไม่สำเร็จ จะไม่มีเสียงยืนยันนี้ (เสียง gyro พร้อมทำงานเป็นอีกเสียงหนึ่ง)

## f_line ออกจากลูปด้วย ADS1115 AIN3

```cpp
f_line(50, 50, 1.5f, ADC(12000.0f), 1);
```

เดินตามเส้นหน้าและอ่าน AIN3 แบบค่าดิบเหมือน `readADC_F()` เมื่อค่าดิบที่อ่านสำเร็จครั้งแรกมากกว่าหรือเท่ากับ 12000 จะออกลูปทันที ไม่รอยืนยันหลายเฟรม ค่าใน `ADC(...)` เป็นค่าดิบ signed -32768 ถึง 32767 ไม่ใช่โวลต์หรือค่า 0–1000 หากถึงเกณฑ์ตั้งแต่เฟรมแรกจะจบได้ทันที หากอ่าน ADS1115 ไม่ได้จะเบรกและรายงาน `ADC_READ_FAILED` หากค่าเกณฑ์ไม่ถูกต้องจะเบรกและรายงาน `INVALID_ADC_THRESHOLD` เมื่อ `stopPull` มากกว่า 0 ใช้เวลาเบรกสวนตามคำสั่ง; 0 คืนโดยไม่สั่งเบรก ยกเว้นกำลังจัดแนวเริ่มต้นซึ่งเบรกทันที การอ่าน AIN3 แบบ single-shot บล็อกระหว่างรอ conversion (สูงสุด 10 ms ต่อครั้งตามไดรเวอร์) จึงอาจลดความถี่การตามเส้น


เซนเซอร์จบของ `b_line()` ไม่รอ 100 ms หรือรอขาว โดยยังยืนยันดำ 2 ใน 3 เฟรม ทั้งนี้ `cl` หรือ `cr` ตัวเดียวต้องผ่านเซนเซอร์หน้านำก่อนตามรายละเอียดด้านล่าง


## เซนเซอร์หน้านำก่อนจบด้วย cl หรือ cr ตัวเดียว

ทั้ง `f_line(..., cl, ...)` และ `b_line(..., cl, ...)` ต้องเห็น `f0` และ `f1` ดำพร้อมกันในเฟรมเดียวก่อน จึงเริ่มอ่านและนับดำ `cl` จากเฟรมถัดไป ส่วน `cr` ต้องเห็น `f14` และ `f15` ดำพร้อมกันก่อน คู่หน้าใช้เกณฑ์ดำสำหรับเซนเซอร์จบ 450/1000 และจำผลตลอดคำสั่ง โดยไม่ต้องรอขาว `cl`/`cr` ยังต้องดำ 2 ใน 3 เฟรม และกฎ stopPull เดิมยังมีผล คู่ `cl & cr` ไม่ใช้เงื่อนไขนำ เซนเซอร์หน้า หลัง และกลางต้องผ่านคาลิเบรตตามที่ใช้ หากคาลิเบรตหน้าไม่พร้อมจะหยุด `INVALID_SENSOR` หากคู่หน้าไม่พบจะไม่จบด้วย cl/cr และยังมี timeout เดิม


## ออกลูปเมื่อพบเส้นด้านซ้ายหรือขวา

```cpp
f_line(30, 30, 0.5f, fl, 1); // เดินหน้า ซ้าย f0–f3
f_line(30, 30, 0.5f, fr, 1); // เดินหน้า ขวา f12–f15
b_line(30, 30, 0.5f, bl, 1); // ถอย ซ้าย b12–b15
b_line(30, 30, 0.5f, br, 1); // ถอย ขวา b0–b3
```

ซ้ายและขวาอ้างอิงตัวหุ่นยนต์ เซนเซอร์ด้านที่เลือกอย่างน้อย 2 จาก 4 ช่องต้องดำ (ค่านอร์มัลต่ำกว่า 500) ในอย่างน้อย 2 จาก 3 เฟรม เริ่มตรวจตั้งแต่เฟรมแรก ไม่ต้องเห็นเส้นกลาง จึงอาจตรวจโค้งที่เข้าริมได้ด้วย; เส้นเอียงที่ไม่ถึงกลุ่มริม เห็นดำช่องเดียว หรือเข้าเกณฑ์เพียงเฟรมเดียวอาจไม่จบ ต้องทดสอบกับสนามจริง เมื่อ `stopPull` เป็น 0 จะคืนทันทีโดยไม่เบรก หากกำลังจัดแนวออกตัวจะเบรกเพื่อไม่ให้หมุนค้าง โหมดเซนเซอร์เดิมไม่เปลี่ยน หากกำหนด fl/fr กับ b_line หรือ bl/br กับ f_line จะหยุดแจ้งค่าไม่ถูกต้อง


## turn แบบตรวจเส้นกลางอัตโนมัติ

```cpp
f_line(30, 30, 0.5f, fl, 1);
turn(tfl, 90, 5);  // หมุนซ้าย หยุดเมื่อพบกลุ่มเส้นใหม่ใกล้กลาง
if (!turn_auto_succeeded()) { motor(1, 1); return; }
```

รองรับ `tfl`, `tfr`, `tcl`, `tcr`, `tl`, `tr` รูปแบบ 3 อาร์กิวเมนต์เป็นโหมดใหม่ ไม่เปลี่ยน `turn(mode, speed, sensor, stopPull)` เดิม หมุนโดยใช้สัดส่วนมอเตอร์ที่ตั้งไว้ หลังเซนเซอร์กลาง f7 และ f8 ขาว 2 เฟรม เริ่มจับกลุ่มดำใหม่และลดสปีดเป็นไม่เกิน 30 และความเร็วค้นหาที่ตั้งไว้ ยืนยันตำแหน่งกลุ่ม error ±10 ใน 2 จาก 3 เฟรม หลังหมุนอย่างน้อย 40 ms และ f0/f15 ต้องขาวทั้งค่าดิบและค่ากรอง แล้วเบรกสวน `stopPull` ms ตามมอเตอร์จริง ตรวจผลด้วย `turn_auto_succeeded()` เมื่อหาไม่พบหรือเซนเซอร์ผิดพลาดจะเบรกและแจ้ง Serial ใช้ timeout ที่ตั้งใน `set_turn_timeout()`

`tfl/tfr` เมื่อคำสั่งก่อนหน้าจบด้วยแยกด้านข้างจะใช้แยกที่พบแล้ว และวิ่งเข้าจุดหมุนตาม `set_turn_overshoot()` ก่อนหมุน; หากไม่มีแยกก่อนหน้าให้ค้นหาคู่ริมก่อน `tcl/tcr` เดินเข้าหาเซนเซอร์กลางตามโหมดก่อนหมุน ส่วน `tl/tr` เริ่มหมุนทันที ค่าและการตอบสนองขึ้นกับสนามจริง หากเส้นเดิมต่อเนื่องผ่านกลางตลอด อาจรอจน timeout ได้
