void sensor()
  {
    const uint8_t channel = 0; // ใช้เซนเซอร์ F0/B0 ในตัวอย่าง
    readADC_F(channel);  // อ่านค่า F0 ที่แปลงเป็น 0–1000
    minADC_F(channel);   // ค่าต่ำสุดของสเกล F0 = 0 เมื่อคาลิเบรตแล้ว
    maxADC_F(channel);   // ค่าสูงสุดของสเกล F0 = 1000 เมื่อคาลิเบรตแล้ว
    readADC_B(channel);  // อ่านค่า B0 ที่แปลงเป็น 0–1000
    minADC_B(channel);   // ค่าต่ำสุดของสเกล B0 = 0 เมื่อคาลิเบรตแล้ว
    maxADC_B(channel);   // ค่าสูงสุดของสเกล B0 = 1000 เมื่อคาลิเบรตแล้ว
    readADC_CL();        // อ่านค่ากลางซ้าย CL ที่แปลงเป็น 0–1000
    minADC_CL();         // ค่าต่ำสุดของสเกล CL = 0 เมื่อคาลิเบรตแล้ว
    maxADC_CL();         // ค่าสูงสุดของสเกล CL = 1000 เมื่อคาลิเบรตแล้ว
    readADC_CR();        // อ่านค่ากลางขวา CR ที่แปลงเป็น 0–1000
    minADC_CR();         // ค่าต่ำสุดของสเกล CR = 0 เมื่อคาลิเบรตแล้ว
    maxADC_CR();         // ค่าสูงสุดของสเกล CR = 1000 เมื่อคาลิเบรตแล้ว

    int32_t aux = readADC_F();  // ADS1115 AIN3: ค่าดิบ (ยังไม่คาลิเบรตเป็น 0–1000)
    if (aux != ADC_VALUE_INVALID) {
      Serial.print("AIN3 raw: "); Serial.println(aux);
    } else {
      Serial.println("AIN3 read failed");
    }

    read_rowSensor();
  }

## คาลิเบรต BNO055 ที่ใช้กับหุ่นยนต์

หลัง `robot_setup()` และก่อนเริ่มภารกิจ วางหุ่นยนต์นิ่ง แล้วเรียก:

```cpp
if (gyro_calibrate(true)) {
  Serial.println("Gyro offset บันทึกและตรวจสอบแล้ว");
} else {
  Serial.println("Gyro calibration ไม่สำเร็จ");
}
```

`gyro_calibrate(true)` บังคับเก็บ gyro offset ใหม่ หากต้องการโหลดค่าเดิมก่อนและเก็บใหม่เฉพาะเมื่อไม่มีค่า ใช้ `gyro_calibrate(false)` ฟังก์ชันหยุดมอเตอร์ เตรียม BNO055 ตรวจตัวอย่างขณะนิ่ง บันทึกและอ่าน EEPROM กลับ จากนั้นตั้งมุมอ้างอิงใหม่ คืน `false` เมื่อไม่พร้อม เคลื่อนระหว่างเก็บ หรือบันทึก/อ่านยืนยันไม่สำเร็จ เฉพาะ BNO055 ไม่รองรับ BNO085

ครั้งต่อไปเมื่อ `gyro_reset()` / `gyro_set()` / `wait_button()` เริ่ม BNO055 ไลบรารีจะพยายามโหลด gyro offset เดิมจาก EEPROM โดยไม่เก็บใหม่อัตโนมัติ ตรวจสถานะด้วย `gyro_offset_calibrated()`; ถ้าไม่มีค่าเดิมยังใช้ไจโรต่อได้ตามปกติ

offset นี้ใช้แก้ค่าอัตราหมุน X/Y/Z ที่ไลบรารีอ่าน รวมถึงตรวจความสมเหตุสมผลของ heading; มุม heading ยังมาจาก fusion ของ BNO055 ไม่ใช่การนำ offset นี้ไปหักจากมุมโดยตรง การบันทึก fusion profile ด้วย `save_gyro_calibration()` เป็นอีกกระบวนการหนึ่ง

ใน `MyMINI_16Pro.ino` มีบรรทัด `gyro_calibrate(true)` ที่คอมเมนต์ไว้ ปลดคอมเมนต์เมื่อจะคาลิเบรตใหม่ แล้วใส่คอมเมนต์กลับหลังบันทึกสำเร็จ เพื่อไม่เขียน EEPROM ซ้ำทุกครั้งที่เปิดเครื่อง

เมื่อคาลิเบรตสำเร็จและตรวจสอบค่าที่บันทึกแล้ว จะมีเสียงเมโลดีสั้น 3 โน้ต เสียงเดียวกันดังเมื่อเข้า `wait_button()` แล้วพบ BNO055 พร้อมค่า gyro offset ที่โหลดไว้ หากไม่พบค่า หรือคาลิเบรตไม่สำเร็จ จะไม่มีเสียงยืนยันนี้ (เสียง gyro พร้อมทำงานเป็นอีกเสียงหนึ่ง)
