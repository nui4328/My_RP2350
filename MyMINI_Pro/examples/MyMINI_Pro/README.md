    turn(tcr, 60, f11, 5);  // ตามเส้นจน cr เห็นดำ แล้วหมุนขวาจน f11 แตะเส้น
    turn(tl,  60, f4,  5);  // หมุนซ้ายทันทีจน f4 แตะเส้น
    turn(tr,  60, f11, 5);  // หมุนขวาทันทีจน f11 แตะเส้น

    turn_gyro(tfr, 60, 90.0f, 20);   // เข้าจุดเลี้ยวแล้วหมุนขวาสู่หัว +90°; 20° เป็นเพดานมุมเผื่อเบรก
    turn_gyro(tfl, 60, -90.0f, 20);  // เข้าจุดเลี้ยวแล้วหมุนซ้ายสู่หัว -90°
    // turn_gyro(tfl, 60, 0.0f, 20); // จากหัว +90° กลับหัว 0° ด้วยการหมุนซ้าย

void sensor()
  {
    const uint8_t channel = 0; // ใช้เซนเซอร์ F0/B0 ในตัวอย่าง
    readADC_F(channel);  // อ่านค่า F0 ที่แปลงเป็น 0–1000
    minADC_F(channel);   // ค่าต่ำสุดของสเกล F0 = 0 เมื่อคาลิเบรตแล้ว
    maxADC_F(channel);   // ค่าสูงสุดของสเกล F0 = 1000 เมื่อคาลิเบรตแล้ว
    readADC_B(channel);  // อ่านค่า B0 ที่แปลงเป็น 0–1000
    minADC_B(channel);   // ค่าต่ำสุดของสเกล B0 = 0 เมื่อคาลิเบรตแล้ว
    maxADC_B(channel);   // ค่าสูงสุดของสเกล B0 = 1000 เมื่อคาลิเบรตแล้ว
    readADC_CL();        // อ่านค่ากลางซ้าย CL ที่แปลงเป็น 0–1000
    minADC_CL();         // ค่าต่ำสุดของสเกล CL = 0 เมื่อคาลิเบรตแล้ว
    maxADC_CL();         // ค่าสูงสุดของสเกล CL = 1000 เมื่อคาลิเบรตแล้ว
    readADC_CR();        // อ่านค่ากลางขวา CR ที่แปลงเป็น 0–1000
    minADC_CR();         // ค่าต่ำสุดของสเกล CR = 0 เมื่อคาลิเบรตแล้ว
    maxADC_CR();         // ค่าสูงสุดของสเกล CR = 1000 เมื่อคาลิเบรตแล้ว
  }

void gyro_mini()
  {
    float yaw, rateX, rateY, rateZ;
    if (!gyro_read(yaw, rateX, rateY, rateZ)) return; // อ่านหัวและอัตราหมุน; อ่านไม่ได้ให้ออก
    Serial.print(F("Yaw before reset: "));
    Serial.println(yaw, 2);
    Serial.print(F("Gyro X/Y/Z (deg/s): "));
    Serial.print(rateX, 2); Serial.print('/');
    Serial.print(rateY, 2); Serial.print('/');
    Serial.println(rateZ, 2);
    if (!gyro_reset() || !gyro_read(yaw, rateX, rateY, rateZ)) return; // รีเซตหัวเป็น 0° แล้วอ่านยืนยัน
    Serial.print(F("Yaw after reset: "));
    Serial.println(yaw, 2);
  }
